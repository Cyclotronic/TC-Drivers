; Agilent / Keysight E364xA DC Power Supplies - TestController driver
;
; Coverage   E3640A, E3641A, E3642A, E3643A, E3644A and E3645A - the
;            single-output, dual-range models (30 W, 50 W and 80 W tiers),
;            under Agilent or Keysight branding. The family postdates the HP
;            name; nothing reports HEWLETT-PACKARD in *IDN?.
;            Both output ranges, voltage and current setpoints with a fine
;            adjust page, overvoltage protection with a trip indicator, bus
;            and immediate triggering, front-panel display text, five stored
;            states, and a regulation readout logged as CC or CV. Five
;            settings pages sit behind a Settings View selector; the output,
;            the range and the two setpoints stay visible above it.
;            There is no overcurrent protection - the E364xA has no CURR:PROT.
;
; Verified   Against an E3641A (firmware 1.8-5.0-1.0) over GPIB, page by page,
;            with the output run into an electronic load for the CV/CC pass.
;            The ceilings and the finest step were read back from the
;            instrument rather than taken from the data sheet: LOW 36.05 V /
;            0.824 A, HIGH 61.80 V / 0.515 A, OVP 1-66 V, finest step
;            0.001127 V / 0.000014 A. On the LOW range the instrument accepts
;            a setpoint above the range ceiling without complaint; the P60V
;            ceiling is enforced. The other five models carry manual-derived
;            range and step figures and still need hardware confirmation.
;            See instrument-notes.md beside this file.
;
; Revision   1.3
; Date       2026-08-28
; Author     Cyclotron
; License    MIT - see LICENSE
;
; Latest version: https://github.com/Cyclotronic/TC-Drivers
; TestController by HKJ: https://lygte-info.dk/project/TestControllerIntro%20UK.html
; Discussion: https://www.eevblog.com/forum/testgear/program-that-can-log-from-many-multimeters/

#driver SCPIx
#port comfixedbaud GPIB
#baudrate 9600N82RD
#eol \n

#gpibReadEol eoi

#interfaceType PS Readout

; Logging queries & value channels
#askValues [mode:Log_All,Log_V_and_I_only]MEAS:VOLT?;[20];MEAS:CURR?;[20];[mode:Log_All]VOLT?;[20];CURR?;[20];STAT:QUES:COND?;[20];[mode:]
#value Voltage V D4 Log_All,Log_V_and_I_only
#value Current A D4 Log_All,Log_V_and_I_only
#value VoltageSet V D4 Log_All
#value CurrentSet A D4 Log_All
#value Regulation - DIGITAL(CC,CV) Log_All
#readingDelay 2
#modeChangeDelay 0.5
#askMode LogModeGet?
#cmdModeLayout 1 2

; SCPI View & Serial/GPIB management
#scpiCmd setView none
:setvar: viewState=inputValue

#scpiCmd getView? none?
:readmath: viewState

#scpiCmd InitView none
:setvar: viewState="Basic_Setup"

#scpiCmd SerialInit #pgm#
if (portType!="GPIB")
  deviceWrite(handle,"SYST:REM");
endif;

#scpiCmd SerialExit #pgm#
if (portType!="GPIB")
  deviceWrite(handle,"SYST:LOC");
endif;

#scpiCmd GpibExit
[LOC]

#scpiCmd SetDisplayText tx DISP:TEXT "(value)"

#scpiCmd SetModeAll none
:setvar: logMode="ALL"

#scpiCmd SetModeVI none
:setvar: logMode="VI"

#scpiCmd LogModeGet? none?
:readmath: logMode

#cmdMode Log_All ALL
SetModeAll

#cmdMode Log_V_and_I_only VI
SetModeVI

#initCmd SetModeAll;SerialInit;InitView;
#finalCmd OUTP OFF;[*OPC];SerialExit;GpibExit
#outputOff OUTP OFF
